def inventory(directory,excluded):
 rows=[]
 for p in sorted(directory.rglob('*')):
  if p.is_file() and p.name not in excluded:
   raw=p.read_bytes();rows.append(dict(path=str(p.relative_to(directory)),bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()))
 return dict(root=str(directory),files=rows,file_count=len(rows),bytes=sum(r['bytes'] for r in rows))
