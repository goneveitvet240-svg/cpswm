from datetime import timedelta
from dataclasses import replace
import json
from test_structure_two_continuous_input import setup, raw_for
from test_natural_vision import fixture_detector,prediction
from test_natural_hands import FakeHands
from test_archive_media_timeline import raw_media
from test_hand_object_evidence import frames
from cpswm.perception_mapping.natural_vision import NaturalVisionEvidenceProducer
from cpswm.perception_mapping.hand_object_evidence import measure_hand_object_evidence
from cpswm.system.reproducibility import content_sha256
_,_,_,transition=setup(False)
t=transition.after.detection_time
raws=[]
for i,(camera,media) in enumerate([('z',1.),('a',1.),('z',0.),('a',0.),('z',1.5),('a',1.5)]):
 raw=raw_media(raw_for(transition,capture=t+timedelta(seconds=30*i),arrival=t+timedelta(seconds=30*i)),media)
 env=raw.envelope(); env=env.model_copy(update={'sensor':env.sensor.model_copy(update={'sensor_id':camera})})
 raws.append(replace(raw,envelope_json=env.model_dump_json()))
d=fixture_detector(raws[0],prediction()); p=NaturalVisionEvidenceProducer(d,FakeHands(d))
for n in range(1,len(raws)+1):
 p.infer(tuple(raws[:n]),cutoff=raws[n-1].envelope().arrival_time)
 evidence=p.hand_object_evidence()
 assert [e.observation_id for e in evidence]==[r.envelope().identity.observation_id for r in raws[:n]]
 restored=NaturalVisionEvidenceProducer(d,FakeHands(d)); restored.restore_state(p.checkpoint_state())
 assert restored.hand_object_evidence()==evidence
 assert restored.interactions()==p.interactions()
print('6 incremental prefixes: 2 cameras, out-of-order archive times, 30-second import gaps, restore equality PASS')
# Timeline trust linkage must reject a checkpoint edit even with a fresh receipt.
before=p.checkpoint_state(); bad=p.checkpoint_state(); raw=bad['prefix'][0]
receipt=json.loads(raw.archive_sampling_json); receipt['sampling_grid_time_seconds']=.25
bad['prefix']=(replace(raw,archive_sampling_json=json.dumps(receipt),capture_receipt_sha256=content_sha256(receipt)),)+bad['prefix'][1:]
try:p.restore_state(bad)
except ValueError:pass
else:raise AssertionError('forged checkpoint accepted')
assert p.checkpoint_state()==before
print('repinned raw timeline cannot reuse old derived checkpoint; atomic rejection PASS')
# Entire receipt copy across observation identities fails before model call.
try:replace(raws[1],archive_sampling_json=raws[0].archive_sampling_json,capture_receipt_sha256=raws[0].capture_receipt_sha256).envelope()
except ValueError:pass
else:raise AssertionError('receipt transplantation accepted')
print('receipt transplantation across observation IDs rejected PASS')
v,h,a=frames()
try:measure_hand_object_evidence(v,h,replace(a,detections=tuple(replace(x,status='CALIBRATED_CONTACT') for x in a.detections)))
except ValueError:pass
else:raise AssertionError('status promotion accepted')
print('forged contact-status promotion rejected PASS')
# Explicit trust limitation: coherently re-authored importer metadata cannot be authenticated without original media.
x=json.loads(raws[0].archive_sampling_json); x['sampling_grid_time_seconds']=1234.
r=replace(raws[0],archive_sampling_json=json.dumps(x),capture_receipt_sha256=content_sha256(x)); assert r.archive_timeline()[1]==1234.
print('coherent new importer receipt accepted: binding, not original-video authenticity (expected scope)')
