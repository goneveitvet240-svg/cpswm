# Recompute every fixed-feature prediction, training standardization and metrics.
A=B/'instance-foreground-20260930/experiment';C=B/'affinity-controls-20261001/experiment'
cr=load(C/'report.json'); base=load(A/'model.json');data=[]
for i in range(96):
 folder=A/f'frames/{i:03d}'
 data.append(tuple(np.load(folder/name,allow_pickle=False) for name in ['features.npy','targets.npy','valid.npy']))
xt=np.concatenate([x for x,y,v in data[:64]]);yt=np.concatenate([y for x,y,v in data[:64]])
close(base['prevalence'],float(np.mean(yt[yt>=0])))
predictions={};all_affinity={};by_house={}
for mode,columns in {'rgb_only':[0,1,2],'geometry_only':[3,4,5,6,7],'combined':list(range(8))}.items():
 wrapper=load(C/f'models/{mode}.json');m=wrapper['inner_checkpoint']
 masked=np.zeros_like(xt);masked[:,columns]=xt[:,columns];fitx=masked[yt>=0]
 close(m['mean'],fitx.mean(0))
 std=fitx.std(0);std[np.ptp(fitx,axis=0)==0]=1
 close(m['scale'],std)
 theta=np.r_[m['weights'],m['bias']];z=np.column_stack(((fitx-m['mean'])/m['scale'],np.ones(len(fitx))))
 grad=z.T@(expit(z@theta)-yt[yt>=0])/len(z)+np.r_[.01*np.asarray(m['weights']),0]
 assert np.max(np.abs(grad))<1e-8
 values=[]
 for i,(x,y,valid) in enumerate(data):
  masked=np.zeros_like(x);masked[:,columns]=x[:,columns]
  pred=expit(((masked-np.asarray(m['mean']))/m['scale'])@m['weights']+m['bias'])
  saved=np.load(C/f'frames/{i:03d}/{mode}.npy',allow_pickle=False)
  close(pred[valid],saved[valid]);assert len(saved)==len(x)
  values.append(pred)
 predictions[mode]=values
predictions['training_frequency_constant']=[np.full(len(y),base['prevalence']) for x,y,v in data]
for mode,values in predictions.items():
 def metrics(indices):
  y=np.concatenate([data[i][1] for i in indices]);p=np.concatenate([values[i] for i in indices]);result={}
  for label,keep in {'all':y>=0,'same_instance':y==1,'different_instance':y==0}.items():
   yy=y[keep];pp=p[keep]
   result[label]={'pairs':len(yy),'bce':float(np.mean(-yy*np.log(pp)-(1-yy)*np.log1p(-pp))) if len(yy) else None,'brier':float(np.mean((pp-yy)**2)) if len(yy) else None}
  return result
 all_affinity[mode]=metrics(range(64,96))
 for cls,r in all_affinity[mode].items():
  expected=cr['by_split']['validation']['methods'][mode][cls]
  assert r['pairs']==expected['pairs'];close(r['bce'],expected['bce']);close(r['brier'],expected['brier'])
 by_house[mode]={str(h):metrics(range((h-1)*8,h*8)) for h in range(1,13)}

  if h>8:
   den['frames'].add(i);den['objects'].add((h,lab['object_id']));den['object_frames'].add((i,lab['object_id']))

for (est,ref),rows in errors.items():
 name=est+'/'+ref;path=(N/'bare/models'/f'{ref}.json') if est=='bare_seed' else P/'models'/f'{name}.json'

# Independent Gaussian/Kalman arithmetic for the four actual Native outputs.
native={};nr=load(N/'bridge/report.json');cor=load(N/'bridge/public-correspondence.json')
for name,result in nr['results'].items():
 m=models[name];est=name.split('/')[0];x=np.asarray(cor['observations'][est]['world_point_m']);R=np.asarray(m['covariance']);z=x-m['bias']
 ll=multivariate_normal.logpdf(z,cov=np.eye(3)+R);unknown=multivariate_normal.logpdf(x,cov=100*np.eye(3))
 logs=np.array([ll,unknown,unknown]);logs-=logsumexp(logs)

 for key,logv in zip(['known','unknown','aggregate'],logs,strict=True):
  close(a['active'][key]['normalized_log_weight'],logv)
  assert math.isclose(a['active'][key]['probability'],float(np.exp(logv)),rel_tol=3e-10,abs_tol=0)
 close(a['active']['known']['information'],precision);close(a['active']['known']['information_vector'],natural)
 for arm in result['arms'].values():
  for key in ['known','unknown','aggregate']:
   expected=result['arms']['no_factor']['active'][key];got=arm['after_retraction'][key]
   for field in expected:close(got[field],expected[field])
  assert arm['duplicate_unchanged'] and arm['raw_packet_unchanged'] and arm['physical_camera_commands']==0
 native[name]={'known_ll':float(ll),'unknown_ll':float(unknown),'known_probability':float(np.exp(logs[0])),'retraction_numeric_match':True}
