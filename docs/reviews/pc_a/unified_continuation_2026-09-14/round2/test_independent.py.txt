def command_for(stream,transition):
    when=transition.after.detection_time+timedelta(seconds=2)
    ids=tuple(r.envelope().identity.observation_id for r in stream.visible_prefix(cutoff=when))
    return stream.prepare_observation(action='Pass',degrees=0,reason='independent probe',source_ids=ids,decision_time=when)

def test_observation_transport_receipt_alias_cannot_rewrite_durable_history(tmp_path):
    _,stream,backend,store,tr=durable_setup(tmp_path/'state.db');deliver(stream,backend,tr)
    cmd=command_for(stream,tr)
    raw=raw_for(tr,capture=cmd.decision_time+timedelta(seconds=1))
    reply=ObservationDelivery(cmd.action_id,(raw,),True,'',cmd.decision_time+timedelta(seconds=1))
    class E:
        def execute(self,c):return reply
    stream.execute_observation(cmd,executor=E())
    object.__setattr__(reply,'success',False)
    object.__setattr__(reply,'error','forged after acceptance')
    assert stream._observation_status[cmd.action_id].success is True
    store.close()

def test_effect_failure_checkpoint_recovery_rejects_reexecute(tmp_path):
    _,stream,backend,store,tr=durable_setup(tmp_path/'state.db');deliver(stream,backend,tr)
    cmd=command_for(stream,tr)
    calls=[]
    class E:
        def execute(self,c):calls.append(c.action_id);raise OSError('effect then lost response')
    with pytest.raises(OSError):stream.execute_observation(cmd,executor=E())
    store.close();store=store_at(tmp_path/'state.db');restored=ContinuousEvidenceInput.resume(store,producer=DurableFixtureProducer())
    own=restored._observation_commands[cmd.action_id][0]
    with pytest.raises(ValueError):restored.execute_observation(own,executor=E())
    assert calls==[cmd.action_id]
    raw=raw_for(tr,capture=cmd.decision_time+timedelta(seconds=1))
    bad=ObservationDelivery(cmd.action_id,(),True,'',cmd.decision_time+timedelta(seconds=1))
    with pytest.raises(ValueError):restored.reconcile_observation(cmd.action_id,bad)
    assert restored._observation_status[cmd.action_id]=='OUTCOME_UNCERTAIN'
    restored.reconcile_observation(cmd.action_id,replace(bad,observations=(raw,)))
    assert len(restored.visible_prefix(cutoff=bad.received_at))==2
    store.close()

def test_actual_p5_effect_reused_after_final_save_failure(tmp_path):
    from test_structure_two_adaptive_runtime import _adaptive_system_and_transition,_ciav_input,_features
    from cpswm.system.structure_two_adaptive_runtime import AdaptiveExecutionContext
    from cpswm.system.structure_two_continuous_input import GroundedTransition
    system,tr=_adaptive_system_and_transition();original=_ciav_input(tr);calls=[]
    class Transport:
        def execute(self,op):
            self.calls.append(op)
            return self.original.realizer(op)
    transport=Transport();transport.calls=calls;transport.original=original
    ciav=replace(original,realizer=transport.execute)
    def builder(sys,item,when,step):return AdaptiveExecutionContext(router_features=_features(sys,step=step,route='P0_FAST_LOCAL'),step_index=step,ciav_input=ciav)
    meta=tr.after.metadata;store=store_at(tmp_path/'p5.db');producer=DurableFixtureProducer()
    stream=ContinuousEvidenceInput(system=system,execution_lane='registered_p5_first',household_id=meta.household_id,session_id=meta.session_id,trace_id=meta.trace_id,producer=producer,state_store=store,context_builder=builder)
    when=ciav.opportunity_time;ids=stream.admit((raw_for(tr),),received_at=when)
    producer.output=GroundedTransition(tr,ids,'independent-fixture','not-real-calibration')
    original_save=store.save
    def fail_after_commit(state):
        if state['fields']['_advanced']:raise OSError('independent simulated final checkpoint loss')
        original_save(state)
    store.save=fail_after_commit
    with pytest.raises(OSError):stream.advance(cutoff=when)
    assert len(calls)==1
    store.close();store=store_at(tmp_path/'p5.db')
    restored=ContinuousEvidenceInput.resume(store,producer=DurableFixtureProducer(),context_builder=builder)
    answer=restored.advance(cutoff=when)
    assert answer.result.path_selection.selected_path_id=='P5_FULL_EAGER'
    assert len(calls)==1 and len(restored.execution_traces())==1
    assert restored._system._assembly_components[0] is restored._system.core
    store.close()

def test_visual_late_recompute_and_config_boundary():
    from test_structure_two_continuous_input import setup
    from test_natural_vision import fixture_detector,prediction
    from cpswm.perception_mapping.natural_vision import NaturalVisionEvidenceProducer
    _,_,_,tr=setup(False);when=tr.after.detection_time
    first=raw_for(tr,capture=when);late=raw_for(tr,capture=when-timedelta(seconds=.5),arrival=when+timedelta(seconds=1))
    detector=fixture_detector(first,prediction());producer=NaturalVisionEvidenceProducer(detector)
    producer.infer((first,),cutoff=when);before=producer.interactions()
    producer.infer((first,late),cutoff=when+timedelta(seconds=1))
    after=producer.interactions()
    assert len(before)==1 and len(after)==2
    assert after[0][0].observation_id==late.envelope().identity.observation_id
    assert producer.frames()[0].observation_id==first.envelope().identity.observation_id
    state=producer.checkpoint_state()
    other=fixture_detector(first,prediction());other._minimum_score=.75
    with pytest.raises(ValueError):NaturalVisionEvidenceProducer(other).restore_state(state)

def test_cross_process_graph_and_late_retraction_changes_next_placement(tmp_path):
    import subprocess,sys,os
    from test_structure_two_continuous_input import old,FixtureWorldExecutor
    p,s,b,store,tr=durable_setup(tmp_path/'state.db')
    for day in p.observed_days()[:12]:deliver(s,b,p.transition_for(day))
    when=p.observed_days()[12].after.detection_time
    first=s.prepare_habit_placement(decision_time=when)
    world=FixtureWorldExecutor(tr.after.metadata);s.execute_placement(first,executor=world)
    bundles=[old._feedback(p,rid) for rid,event in p.system.core._committed_events.items() if event.location_id==first.location_id]
    expected=p.system.adaptive_router_state_sha256();store.close()
    script='''from pathlib import Path
import sys
from cpswm.system.structure_two_continuous_input import ContinuousEvidenceInput
from cpswm.system.continuous_state_store import ContinuousStateStore
s=ContinuousStateStore(Path(sys.argv[1]),source_identity="a"*64,dependency_identity="b"*64)
g=s.load()["fields"]["_system"]
assert g._assembly_components[0] is g.core
print(g.adaptive_router_state_sha256())
s.close()
'''
    result=subprocess.run([sys.executable,'-c',script,str(tmp_path/'state.db')],capture_output=True,text=True,env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1'})
    assert result.returncode==0,result.stderr
    assert result.stdout.strip()==expected
    store=store_at(tmp_path/'state.db');r=ContinuousEvidenceInput.resume(store,producer=DurableFixtureProducer())
    for i,(fb,bind,lm) in enumerate(bundles):r.consume_feedback(feedback=fb,binding=bind,likelihood_model=lm,received_at=when+timedelta(seconds=i+2),policy=old.RETRACTION_POLICY)
    second=r.prepare_habit_placement(decision_time=when+timedelta(minutes=1));assert first.location_id!=second.location_id
    r.execute_placement(second,executor=world);assert world.calls==2
    assert r._system.core.verify_hybrid_full_rerun_equivalence().equivalent
    store.close()
