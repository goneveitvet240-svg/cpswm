from datetime import timedelta
from dataclasses import replace
from uuid import uuid4
from test_structure_two_continuous_input import setup, raw_for
from test_natural_vision import fixture_detector, prediction
from test_natural_hands import FakeHands
from test_hand_object_evidence import frames
from cpswm.perception_mapping.natural_vision import NaturalVisionEvidenceProducer
from cpswm.perception_mapping.hand_object_evidence import measure_hand_object_evidence
_,_,_,transition=setup(False)
t=transition.after.detection_time
newer=raw_for(transition,capture=t,arrival=t)
older=raw_for(transition,capture=t-timedelta(seconds=.5),arrival=t+timedelta(seconds=1))
d=fixture_detector(newer,prediction())
p=NaturalVisionEvidenceProducer(d,FakeHands(d))
p.infer((newer,),cutoff=t)
p.infer((newer,older),cutoff=t+timedelta(seconds=1))
print('valid late-prefix accepted:',len(p.frames()))
for name,prod in [('current',p),('restored',NaturalVisionEvidenceProducer(d,FakeHands(d)))]:
    if name=='restored': prod.restore_state(p.checkpoint_state())
    try: print(name,prod.hand_object_evidence())
    except ValueError as e: print(name,'BUG:',str(e))
v,h,a=frames()
forged=replace(a,detections=tuple(replace(x,track_id=uuid4(),status='CALIBRATED_CONTACT') for x in a.detections))
r=measure_hand_object_evidence(v,h,forged)
print('forged association status copied:',r.measurements[0].object_association_status,'overall:',r.status)
