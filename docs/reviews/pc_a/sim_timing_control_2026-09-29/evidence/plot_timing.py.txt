import os,sys,json
from pathlib import Path
import numpy as np
os.environ.setdefault('MPLCONFIGDIR','/private/tmp/cpswm-timing-matplotlib')
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
root=Path('/private/tmp/cpswm-pc-a-sim-timing-control-20260929')
sys.path[:0]=[str(root/'src'),str(root/'tools'),str(root/'tests')]
from run_simulator_timing import plan
from run_simulator_repeatability import image_delta,geometry_delta
out=Path(__file__).parent
captures=out/'attempt02/captures'
assert json.loads((captures/'verification.json').read_text())['frames']==384
fig,axes=plt.subplots(4,2,figsize=(11,11),layout='constrained')
colors=dict(auto='#65758B',frozen='#008A82',stepped='#CC633D')
data=[]
for c in plan():
 row=(0 if c['site']=='north' else 2)+(0 if c['size']==320 else 1)
 p=captures/c['name']/'unity-logs/evaluator_only/sdk-events'
 rgb0=np.load(p/'001-rgb.npy');mask0=np.load(p/'001-mask.npy')
 meta0=json.loads((p/'001.json').read_text())['metadata']
 x=list(range(1,34));rgb=[];movement=[]
 for i in x:
  a=np.load(p/f'{i:03d}-rgb.npy');m=np.load(p/f'{i:03d}-mask.npy')
  meta=json.loads((p/f'{i:03d}.json').read_text())['metadata']
  rgb.append(image_delta(rgb0,a,mask0|m)['mean_abs_byte'])
  movement.append(1000*geometry_delta(meta0,meta)['max_object_translation'])
 line=dict(color=colors[c['mode']],linewidth=1.8,alpha=.8,linestyle='-' if c['repeat']==0 else '--',label=c['mode'] if c['repeat']==0 else None)
 axes[row,0].plot(x,rgb,**line);axes[row,1].plot(x,movement,**line)
 axes[row,0].set_title(f"{c['site'].capitalize()} / {c['size']} px - RGB change")
 axes[row,1].set_title(f"{c['site'].capitalize()} / {c['size']} px - object displacement")
 data.append(dict(cell=c,sdk_event_index=x,mean_absolute_byte_from_prepare=rgb,max_coordinate_displacement_mm_from_prepare=movement))
for row in range(4):
 for col in range(2):
  ax=axes[row,col];ax.grid(alpha=.18);ax.set_xlabel('SDK event index (not physical seconds)')
  ax.set_xlim(1,33);ax.set_xticks([1,3,11,19,27,33]);ax.axvspan(19,33,color='#BEC6D0',alpha=.12)
  ax.spines[['top','right']].set_visible(False)
 axes[row,0].set_ylabel('Mean absolute RGB byte difference')
 axes[row,1].set_ylabel('Maximum coordinate change (mm)')
axes[0,0].legend(loc='lower right',frameon=False,ncols=3)
fig.suptitle('Fixed-camera timing interventions: all 24 runs\nSolid/dashed: two launches. Shading: predeclared final 8 public frames.',fontsize=13)
fig.savefig(out/'timing-control.png',dpi=160)
fig.savefig(out/'timing-control.svg')
(out/'plot-data.json').write_text(json.dumps(data,indent=2)+'\n')
print(out/'timing-control.png')
